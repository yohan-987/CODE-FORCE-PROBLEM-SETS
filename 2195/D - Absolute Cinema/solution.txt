t = int(input().strip())
for k in range(t):
    n = int(input())
    f = list(map(int, input().split()))
    S = (f[0] + f[-1]) // (n - 1)
    a = [0] * n
    mid_sum = 0 
    for i in range(1, n - 1):
        ai = (f[i - 1] + f[i + 1] - 2 * f[i]) // 2
        a[i] = ai
        mid_sum += ai
    a[0] = (f[1] - f[0] + S) // 2
    a[-1] = S - (a[0] + mid_sum)
    print(*a)
