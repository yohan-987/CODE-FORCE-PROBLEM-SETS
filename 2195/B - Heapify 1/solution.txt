def root(x):
    while x % 2 == 0:
        x //= 2
    return x
t = int(input())
for i in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    f = True
    for k in range(1, n + 1):
        if root(k) != root(a[k - 1]):
            f = False
            break
    print("YES" if f else "NO")
