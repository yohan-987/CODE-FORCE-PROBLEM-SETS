t=int(input())
for i in range(t):
    n=int(input())
    a=list(map(int,input().split()))
    c=[]
    for j in range(n):
        if j%2==0:
            c.append("R")
        else:
            c.append("B")
    b=sorted(a)
    
    for k in range(n-1):
       x=True
       if c[a.index(b[k])] == c[a.index(b[k+1])]:
           
           x=False
           break
       
    if x:
        print("YES")
    else:
        print("NO")
